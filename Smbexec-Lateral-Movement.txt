rule smbexec_lateral_movement {
  
events:
    $e.metadata.event_type = "PROCESS_LAUNCH"
    $e.target.process.file.full_path = /\\cmd\.exe$/ nocase
    $e.target.process.command_line = /\/Q/ nocase
    $e.target.process.command_line = /\/c/ nocase
    $e.target.process.command_line = /echo/ nocase
    $e.target.process.command_line = /C:\\Windows\\TEMP\\/ nocase
    $e.target.process.command_line = /\.bat/ nocase
    $e.target.process.command_line = /del/ nocase

  condition:
    $e
}
